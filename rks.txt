import pandas as pd
import plotly.express as px
import requests

# 1. Wczytanie danych (utf-8-sig usuwa ewentualny znak BOM z początku pliku)
sciezka = "Kopia Kopia ceny_mieszkan_polska_2026 (1).csv"
df = pd.read_csv(sciezka, encoding="utf-8-sig")

# 2. Mapowanie miast na województwa (nazwy małymi literami jak w pliku GeoJSON)
miasto_do_woj = {
    "Warszawa": "mazowieckie",
    "Radom": "mazowieckie",
    "Kraków": "małopolskie",
    "Gdańsk": "pomorskie",
    "Gdynia": "pomorskie",
    "Wrocław": "dolnośląskie",
    "Poznań": "wielkopolskie",
    "Szczecin": "zachodniopomorskie",
    "Katowice": "śląskie",
    "Częstochowa": "śląskie",
    "Sosnowiec": "śląskie",
    "Gliwice": "śląskie",
    "Zabrze": "śląskie",
    "Rzeszów": "podkarpackie",
    "Lublin": "lubelskie",
    "Łódź": "łódzkie",
    "Opole": "opolskie",
    "Białystok": "podlaskie",
    "Bydgoszcz": "kujawsko-pomorskie",
    "Toruń": "kujawsko-pomorskie",
    "Olsztyn": "warmińsko-mazurskie",
    "Kielce": "świętokrzyskie",
    "Gorzów Wielkopolski": "lubuskie",
}

df["Województwo"] = df["Miasto"].map(miasto_do_woj)

# 3. Agregacja po województwach (średnia cena + lista miast do podglądu po najechaniu)
woj_df = (
    df.groupby("Województwo", as_index=False)
    .agg(
        srednia_cena=("Średnia cena za m² (PLN)", "mean"),
        miasta=("Miasto", lambda x: ", ".join(x)),
    )
)
woj_df["srednia_cena"] = woj_df["srednia_cena"].round(0)

# 4. Pobranie granic województw Polski (GeoJSON)
geojson_url = (
    "https://raw.githubusercontent.com/ppatrzyk/polska-geojson/"
    "master/wojewodztwa/wojewodztwa-min.geojson"
)
polska_geojson = requests.get(geojson_url).json()

# 5. Generowanie kartogramu
fig = px.choropleth(
    woj_df,
    geojson=polska_geojson,
    locations="Województwo",
    featureidkey="properties.nazwa",
    color="srednia_cena",
    color_continuous_scale="YlOrRd",
    hover_name="Województwo",
    hover_data={"Województwo": False, "srednia_cena": ":.0f", "miasta": True},
    labels={
        "srednia_cena": "Średnia cena (PLN/m²)",
        "miasta": "Uwzględnione miasta",
    },
    title="Średnie ceny mieszkań w Polsce wg województw (Październik 2026)",
)

fig.update_geos(fitbounds="locations", visible=False)
fig.update_layout(margin={"r": 0, "t": 50, "l": 0, "b": 0})

# Zapis do pliku HTML i otwarcie w przeglądarce
fig.write_html("kartogram_ceny_mieszkan_2026.html")
fig.show()
